-- Создаем базу данных
CREATE DATABASE Library;

-- Используем созданную базу данных
USE Library;

-- Таблица для хранения информации о читателях
CREATE TABLE Readers (
    reader_id INT AUTO_INCREMENT PRIMARY KEY,  -- уникальный ID читателя (нередактируемый)
    full_name VARCHAR(255) NOT NULL            -- ФИО читателя
);

-- Таблица для хранения информации о книгах
CREATE TABLE Books (
    book_id INT AUTO_INCREMENT PRIMARY KEY,    -- уникальный ID книги
    author_name VARCHAR(255) NOT NULL,         -- ФИО автора книги
    book_title VARCHAR(255) NOT NULL,          -- Наименование книги
    total_quantity INT NOT NULL                -- Общее количество экземпляров в библиотеке
);

-- Таблица для хранения информации о выданных книгах
CREATE TABLE Issued_Books (
    issue_id INT AUTO_INCREMENT PRIMARY KEY,   -- уникальный ID выдачи
    reader_id INT,                             -- ID читателя, которому выдана книга
    book_id INT,                               -- ID выданной книги
    FOREIGN KEY (reader_id) REFERENCES Readers(reader_id) ON DELETE CASCADE,
    FOREIGN KEY (book_id) REFERENCES Books(book_id) ON DELETE CASCADE
);
